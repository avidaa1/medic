function lev(a,b){
  if(Math.abs(a.length-b.length)>2)return 9;
  const m=a.length,n=b.length,d=Array.from({length:m+1},(_,i)=>[i,...Array(n).fill(0)]);
  for(let j=0;j<=n;j++)d[0][j]=j;
  for(let i=1;i<=m;i++)for(let j=1;j<=n;j++)
    d[i][j]=Math.min(d[i-1][j]+1,d[i][j-1]+1,d[i-1][j-1]+(a[i-1]===b[j-1]?0:1));
  return d[m][n];
}

  const t=document.createElement('div');t.className='typing-indicator';t.id='typing';
  t.innerHTML='<span></span><span></span><span></span>';
  const m=document.getElementById('chat-msgs');m.appendChild(t);m.scrollTop=9999;return t;
}
let chatOpen=false;
function toggleChat(){
  chatOpen=!chatOpen;
  document.getElementById('chat-box').classList.toggle('open',chatOpen);
  if(chatOpen&&document.getElementById('chat-msgs').children.length===0){
    setTimeout(()=>addMsg('MEDIC online \u2014 built from this page, runs in your browser. Ask about a pathogen or drug class.','ai'),350);
  }
  if(chatOpen)document.getElementById('chat-input').focus();
}
function chatSend(){
  const inp=document.getElementById('chat-input');const msg=inp.value.trim();if(!msg)return;
  addMsg(msg,'user');inp.value='';
  if(window.rabbitThink)window.rabbitThink();
  const typing=showTyping();
  setTimeout(()=>{
    typing.remove();
    const res=chatMatch(msg);
    if(!res.handled)addMsg(res.text,'ai',res.link);
    if(window.rabbitRespond)window.rabbitRespond();
    setTimeout(()=>{if(window.rabbitIdle)window.rabbitIdle();},3000);
  },380+Math.random()*340);
}
function quickAsk(q){document.getElementById('chat-input').value=q;chatSend();toggleChat();if(!chatOpen)toggleChat();}
document.getElementById('chat-input').addEventListener('keydown',e=>{if(e.key==='Enter')chatSend();});
